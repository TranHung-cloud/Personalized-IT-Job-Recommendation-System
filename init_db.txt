BEGIN;

CREATE EXTENSION IF NOT EXISTS pgcrypto;

-- =========================================================
-- 1. COMPANY
-- =========================================================

CREATE TABLE company (
    co_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    co_name VARCHAR(255) NOT NULL,
    co_name_norm VARCHAR(255),
    co_logo_path TEXT
);

CREATE INDEX idx_company_name_norm
    ON company(co_name_norm);


-- =========================================================
-- 2. ROLE & ACCOUNT
-- =========================================================

CREATE TABLE role (
    role_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    role_name VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE account (
    acc_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    role_id UUID REFERENCES role(role_id),
    acc_email VARCHAR(320) NOT NULL,
    acc_password_hash TEXT NOT NULL,
    acc_enabled BOOLEAN NOT NULL DEFAULT FALSE,
    acc_created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    acc_deleted_at TIMESTAMPTZ,

    CHECK (
        acc_deleted_at IS NULL
        OR acc_deleted_at >= acc_created_at
    )
);

-- =========================================================
-- 3. CANONICAL DICTIONARIES
-- =========================================================

CREATE TABLE position (
    pos_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    pos_name VARCHAR(255) NOT NULL UNIQUE
);

CREATE TABLE work_mode (
    wm_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    wm_name VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE location (
    loc_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    loc_name VARCHAR(255) NOT NULL UNIQUE
);

CREATE TABLE skill (
    sk_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    sk_name VARCHAR(255) NOT NULL UNIQUE
);


-- =========================================================
-- 4. RAW DICTIONARIES
-- Each raw value maps to zero or one canonical entity.
-- =========================================================

CREATE TABLE position_raw (
    pos_raw_name VARCHAR(255) PRIMARY KEY,
    pos_raw_count BIGINT NOT NULL DEFAULT 1
        CHECK (pos_raw_count >= 1),
    pos_raw_status VARCHAR(30) NOT NULL
        CHECK (
            pos_raw_status IN (
                'pending', 'mapped', 'ignored', 'review'
            )
        ),
    pos_id UUID REFERENCES position(pos_id),

    CHECK (
        pos_raw_status <> 'mapped'
        OR pos_id IS NOT NULL
    )
);

CREATE TABLE work_mode_raw (
    wm_raw_name VARCHAR(255) PRIMARY KEY,
    wm_raw_count BIGINT NOT NULL DEFAULT 1
        CHECK (wm_raw_count >= 1),
    wm_raw_status VARCHAR(30) NOT NULL
        CHECK (
            wm_raw_status IN (
                'pending', 'mapped', 'ignored', 'review'
            )
        ),
    wm_id UUID REFERENCES work_mode(wm_id),

    CHECK (
        wm_raw_status <> 'mapped'
        OR wm_id IS NOT NULL
    )
);

CREATE TABLE location_raw (
    loc_raw_name VARCHAR(255) PRIMARY KEY,
    loc_raw_count BIGINT NOT NULL DEFAULT 1
        CHECK (loc_raw_count >= 1),
    loc_raw_status VARCHAR(30) NOT NULL
        CHECK (
            loc_raw_status IN (
                'pending', 'mapped', 'ignored', 'review'
            )
        ),
    loc_id UUID REFERENCES location(loc_id),

    CHECK (
        loc_raw_status <> 'mapped'
        OR loc_id IS NOT NULL
    )
);

CREATE TABLE skill_raw (
    sk_raw_name VARCHAR(255) PRIMARY KEY,
    sk_raw_count BIGINT NOT NULL DEFAULT 1
        CHECK (sk_raw_count >= 1),
    sk_raw_status VARCHAR(30) NOT NULL
        CHECK (
            sk_raw_status IN (
                'pending', 'mapped', 'ignored', 'review'
            )
        ),
    sk_id UUID REFERENCES skill(sk_id),

    CHECK (
        sk_raw_status <> 'mapped'
        OR sk_id IS NOT NULL
    )
);



-- =========================================================
-- 5. SITE & SOURCE
-- One Site has many Sources.
-- One Source can describe many Jobs.
-- =========================================================

CREATE TABLE site (
    site_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    site_name VARCHAR(255) NOT NULL UNIQUE
);

CREATE TABLE source (
    src_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    site_id UUID REFERENCES site(site_id),
    src_url TEXT NOT NULL,
    src_crawled_at TIMESTAMPTZ,
    src_created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    src_raw_key TEXT,
    src_status VARCHAR(255) CHECK (src_status IN ('pending', 'crawled', 'error', 'duplicate'))
    pos_raw_name VARCHAR(255)
        REFERENCES position_raw(pos_raw_name),
    wm_raw_name VARCHAR(255)
        REFERENCES work_mode_raw(wm_raw_name)
);

CREATE TABLE source_skill_raw (
    src_id UUID NOT NULL
        REFERENCES source(src_id) ON DELETE CASCADE,

    sk_raw_name VARCHAR(255) NOT NULL
        REFERENCES skill_raw(sk_raw_name),

    PRIMARY KEY (src_id, sk_raw_name)
);

CREATE TABLE source_location_raw (
    src_id UUID NOT NULL
        REFERENCES source(src_id) ON DELETE CASCADE,

    loc_raw_name VARCHAR(255) NOT NULL
        REFERENCES location_raw(loc_raw_name),

    PRIMARY KEY (src_id, loc_raw_name)
);

CREATE INDEX idx_source_skill_raw_name
    ON source_skill_raw(sk_raw_name);

CREATE INDEX idx_source_location_raw_name
    ON source_location_raw(loc_raw_name);

-- =========================================================
-- 6. JOB
-- Job contains its core attributes.
-- Position, work mode, location and skill are linked below.
-- =========================================================

CREATE TABLE job (
    job_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    co_id UUID REFERENCES company(co_id),
    src_id UUID REFERENCES source(src_id),
    job_title VARCHAR(500) NOT NULL,
    job_title_norm VARCHAR(500),
    job_exp_min INTEGER
        CHECK (job_exp_min >= 0),
    job_exp_max INTEGER
        CHECK (job_exp_max >= 0),
    job_sal_min INTEGER
        CHECK (job_sal_min >= 0),
    job_sal_max INTEGER
        CHECK (job_sal_max >= 0),
    job_sal_nego BOOLEAN NOT NULL DEFAULT FALSE,
    job_desc TEXT,
    job_req TEXT,
    job_benefit TEXT,
    job_expired_at TIMESTAMPTZ,
    job_posted_at TIMESTAMPTZ,
    
    pos_id UUID REFERENCES position(pos_id),
    wm_id UUID REFERENCES work_mode(wm_id),
    qual_id UUID REFERENCES qualifications(qual_id),
    CHECK (
        job_exp_min IS NULL
        OR job_exp_max IS NULL
        OR job_exp_min <= job_exp_max
    ),

    CHECK (
        job_sal_min IS NULL
        OR job_sal_max IS NULL
        OR job_sal_min <= job_sal_max
    )
);

CREATE INDEX idx_job_company
    ON job(co_id);

CREATE INDEX idx_job_source
    ON job(src_id);

CREATE INDEX idx_job_posted_at
    ON job(job_posted_at);

CREATE INDEX idx_job_title_norm
    ON job(job_title_norm);


-- =========================================================
-- 7. JOB RELATIONSHIPS
-- Many-to-many relationships with canonical dictionaries.
-- =========================================================

CREATE TABLE job_location (
    job_id UUID NOT NULL
        REFERENCES job(job_id) ON DELETE CASCADE,

    loc_id UUID NOT NULL
        REFERENCES location(loc_id),

    PRIMARY KEY (job_id, loc_id)
);

CREATE INDEX idx_job_location_loc_id
    ON job_location(loc_id);


CREATE TABLE job_skill (
    job_id UUID NOT NULL
        REFERENCES job(job_id) ON DELETE CASCADE,

    sk_id UUID NOT NULL
        REFERENCES skill(sk_id),

    PRIMARY KEY (job_id, sk_id)
);

CREATE INDEX idx_job_skill_sk_id
    ON job_skill(sk_id);


-- =========================================================
-- 8. PROFILE
-- Default salary and optional primary preferences.
-- =========================================================

CREATE TABLE profile (
    pf_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    acc_id UUID NOT NULL UNIQUE
        REFERENCES account(acc_id) ON DELETE CASCADE,

    pf_expt_sal_min INTEGER
        CHECK (pf_expt_sal_min >= 0),
);


-- =========================================================
-- 9. PROFILE SKILLS & PREFERENCES
-- =========================================================

CREATE TABLE knows_skill (
    pf_id UUID NOT NULL
        REFERENCES profile(pf_id) ON DELETE CASCADE,

    sk_id UUID NOT NULL
        REFERENCES skill(sk_id),

    PRIMARY KEY (pf_id, sk_id)
);

CREATE TABLE targets_location (
    pf_id UUID NOT NULL
        REFERENCES profile(pf_id) ON DELETE CASCADE,

    loc_id UUID NOT NULL
        REFERENCES location(loc_id),

    PRIMARY KEY (pf_id, loc_id)
);

CREATE TABLE prefers_work_mode (
    pf_id UUID NOT NULL
        REFERENCES profile(pf_id) ON DELETE CASCADE,

    wm_id UUID NOT NULL
        REFERENCES work_mode(wm_id),

    PRIMARY KEY (pf_id, wm_id)
);

CREATE TABLE targets_position (
    pf_id UUID NOT NULL
        REFERENCES profile(pf_id) ON DELETE CASCADE,

    pos_id UUID NOT NULL
        REFERENCES position(pos_id),

    PRIMARY KEY (pf_id, pos_id)
);

-- =========================================================
-- 10. QUALIFICATIONS & DEGREE
-- =========================================================

CREATE TABLE qualifications (
    qual_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    qual_name VARCHAR(255) NOT NULL UNIQUE,
    qual_level INT
);

CREATE TABLE degrees (
    deg_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    pf_id UUID NOT NULL
        REFERENCES profile(pf_id) ON DELETE CASCADE,

    qual_id UUID REFERENCES qualifications(qual_id),

    deg_major VARCHAR(255),
    deg_institution VARCHAR(255),
    deg_issued_at DATE,
    deg_signed_by VARCHAR(255),
    deg_path_img TEXT
);

CREATE INDEX idx_degrees_profile
    ON degrees(pf_id);

CREATE INDEX idx_degrees_qualification
    ON degrees(qual_id);


-- =========================================================
-- 11. WORK EXPERIENCE
-- Use a surrogate primary key so start date may be NULL.
-- =========================================================

CREATE TABLE work_experience (
    pf_id UUID NOT NULL
        REFERENCES profile(pf_id),

    co_id UUID NOT NULL
        REFERENCES company(co_id),

    pos_id UUID NOT NULL
        REFERENCES position(pos_id),

    month_start DATE NOT NULL,
    month_end DATE,
    
    primary key(pf_id, co_id, pos_id, month_start),
    CHECK (
        month_end IS NULL
        OR month_start <= month_end
    )
);

CREATE INDEX idx_work_experience_profile
    ON work_experience(pf_id);

CREATE INDEX idx_work_experience_company
    ON work_experience(co_id);

CREATE INDEX idx_work_experience_position
    ON work_experience(pos_id);


-- =========================================================
-- 12. USER INTERACTION
-- =========================================================

CREATE TABLE user_interaction (

    acc_id UUID NOT NULL
        REFERENCES account(acc_id) ON DELETE CASCADE,
    job_id UUID NOT NULL
        REFERENCES job(job_id) ON DELETE CASCADE,
    action VARCHAR(50) NOT NULL
        CHECK (action IN ('interest', 'ignore')),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    PRIMARY KEY(acc_id, job_id)
);

CREATE INDEX idx_interaction_account
    ON user_interaction(acc_id);

CREATE INDEX idx_interaction_job
    ON user_interaction(job_id);

CREATE INDEX idx_interaction_created_at
    ON user_interaction(created_at);

CREATE INDEX idx_interaction_action
    ON user_interaction(action);

COMMIT;
